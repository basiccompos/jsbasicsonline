1  REM RECAMAN'S SEQUENCE (DIAGONAL VIEW)
10  GOSUB 9000: REM SETUP
20  GOSUB 100: REM NEXT STEP
30  GOSUB 200: REM WILL STEP BACKWARDS TO A POSITION NOT VISITED OR FORWARD INSTEAD?
40  IF NP > MP THEN  END : REM END IF MAX POSITION EXCEEDED
50  GOSUB 300: REM DRAW ARC TO NEW POSITION
60  GOSUB 1100: REM MOVE TO NEW POSITION
70  GOTO 20
100  REM ** NEXT STEP
110 S = S + 1: REM STEP SIZE
120 AP =  NOT AP: REM ALTERNATE ARC PHASE (ABOVE/BELOW AXIS)
199  RETURN 
200  REM ** WILL STEP BACKWARDS TO A POSITION NOT VISITED OR FORWARD INSTEAD?
210 NP = P - S: REM NEW POSITION: TRY TO STEP BACKWARDS FIRST
220  IF NP >  = 0 THEN  IF  NOT V(NP) THEN  RETURN : REM BACKWARDS OK
230 NP = P + S: REM BACKWARDS FAILED, STEP FORWARD INSTEAD
299  RETURN 
300  REM ** DRAW ARC TO NEW POSITION
310  IF P > IS AND NP > IS THEN  RETURN : REM ARC BEYOND SCREEN ENTIRELY, NOTHING TO DRAW
320  GOSUB 400: REM CALCULATE ARC PARAMETERS
330  FOR A = A0 TO A1 STEP SI * AI
340  IF  SGN (A - C1) = SI THEN A = C2:C1 = C3:C2 = C4:C3 = SI * BIG: GOTO 380: REM ANGLE A HAS ENTERED A CLIPPING ZONE: SKIP TO END OF ZONE (BACK TO SCREEN) AND PREPARE FOR NEXT ZONE
350 X =  INT (CX +  COS (A) * R):Y =  INT (CY -  SIN (A) * R): REM POINT ON THE ARC
370  Pset X,Y
380  NEXT A
399  RETURN 
400  REM ** CALCULATE ARC PARAMETERS
410 CP = (P + NP) / 2: REM CENTER POSITION
420 CX = XA + CP * SX:CY = YB - CP * SY: REM CENTER COORDINATES
430 R =  ABS (P - NP) / 2 * SC: REM RADIUS IN PIXELS
440 AI = PI / 4 / R: REM ANGLE INCREMENT
500  REM CALCULATE PARAMETERS BASED ON DIRECTION AND PHASE
510  IF NP > P GOTO 800: REM STEP FORWARD?
520  IF AP GOTO 700: REM ARC ABOVE AXIS?
600  REM STEP BACKWARDS, ARC BELOW AXIS
610 A0 = AA - 0:A1 = AA - PI:SI =  - 1: REM START ANGLE, END ANGLE, SIGN OF ANGLE INCREMENT
620 C1 =  - BIG:C3 =  - BIG: REM NO CLIPPING ZONE INITIALLY
630  IF R > YB - CY THEN T =  FN AC((YB - CY) / R):C1 =  - HP + T:C2 =  - HP - T: REM CLIP ARC AT BOTTOM BORDER
640  IF R > XB - CX THEN C3 = C1:C4 = C2:T =  FN AC((XB - CX) / R):C1 = 0 + T:C2 = 0 - T: REM CLIP ARC AT RIGHT BORDER (FIRST)
650  IF C3 <  >  - BIG THEN  IF C2 <  = C3 THEN C2 = C4:C3 =  - BIG: REM TWO ZONES EXIST AND OVERLAP? MERGE INTO ONE
660  RETURN 
700  REM STEP BACKWARDS, ARC ABOVE AXIS
710 A0 = AA + 0:A1 = AA + PI:SI =  + 1: REM START ANGLE, END ANGLE, SIGN OF ANGLE INCREMENT
720 C1 =  + BIG:C3 =  + BIG: REM NO CLIPPING ZONE INITIALLY
730  IF R > CX - XA THEN T =  FN AC((CX - XA) / R):C1 = PI - T:C2 = PI + T: REM CLIP ARC AT LEFT BORDER
740  IF R > CY - YA THEN C3 = C1:C4 = C2:T =  FN AC((CY - YA) / R):C1 = HP - T:C2 = HP + T: REM CLIP ARC AT TOP BORDER (FIRST)
750  IF C3 <  >  + BIG THEN  IF C2 >  = C3 THEN C2 = C4:C3 =  + BIG: REM TWO ZONES EXIST AND OVERLAP? MERGE INTO ONE
760  RETURN 
800  REM STEP FORWARD
810  IF AP GOTO 1000: REM ARC ABOVE AXIS?
900  REM STEP FORWARD, ARC BELOW AXIS
910 A0 = AA - PI:A1 = AA - 0:SI =  + 1: REM START ANGLE, END ANGLE, SIGN OF ANGLE INCREMENT
920 C1 =  + BIG:C3 =  + BIG: REM NO CLIPPING ZONE INITIALLY
930  IF R > XB - CX THEN T =  FN AC((XB - CX) / R):C1 = 0 - T:C2 = 0 + T: REM CLIP ARC AT RIGHT BORDER
940  IF R > YB - CY THEN C3 = C1:C4 = C2:T =  FN AC((YB - CY) / R):C1 =  - HP - T:C2 =  - HP + T: REM CLIP ARC AT BOTTOM BORDER (FIRST)
950  IF C3 <  >  + BIG THEN  IF C2 >  = C3 THEN C2 = C4:C3 =  + BIG: REM TWO ZONES EXIST AND OVERLAP? MERGE INTO ONE
960  RETURN 
1000  REM STEP FORWARD, ARC ABOVE AXIS
1010 A0 = AA + PI:A1 = AA + 0:SI =  - 1: REM START ANGLE, END ANGLE, SIGN OF ANGLE INCREMENT
1020 C1 =  - BIG:C3 =  - BIG: REM NO CLIPPING ZONE INITIALLY
1030  IF R > CY - YA THEN T =  FN AC((CY - YA) / R):C1 = HP + T:C2 = HP - T: REM CLIP ARC AT TOP BORDER
1040  IF R > CX - XA THEN C3 = C1:C4 = C2:T =  FN AC((CX - XA) / R):C1 = PI + T:C2 = PI - T: REM CLIP ARC AT LEFT BORDER (FIRST)
1050  IF C3 <  >  - BIG THEN  IF C2 <  = C3 THEN C2 = C4:C3 =  - BIG: REM TWO ZONES EXIST AND OVERLAP? MERGE INTO ONE
1060  RETURN 
1100  REM ** MOVE TO NEW POSITION
1110 P = NP: REM UPDATE CURRENT POSITION = NEW POSITION
1120 V(P) = TRUE: REM MARK CURRENT POSITION AS VISITED
1199  RETURN 
9000  REM ** SETUP
9010  REM CONSTANTS
9020 FALSE = 0:TRUE =  NOT FALSE
9030 PIr = 3.14159265
9040 HP = PI / 2: REM HALF PI
9050 BIG = 1E + 37: REM LARGE NUMBER USED AS INFINITY SENTINEL
9100  REM USR FUNCTION
9110 rem DEF Fn AC(X) = HP -  ATN (X /  SQR (1 - X * X)): REM ARCCOSINE(X)
9200  REM SEQUENCE PARAMETERS
9210 MP = 1000: REM MAX POSITION
9220  DIM V(MP): REM VISITED POSITIONS (INITIALLY NONE)
9230 P = 0: REM CURRENT POSITION (INITIALLY 0)
9240 V(P) = TRUE: REM MARK INITIAL POSITION AS VISITED
9250 AP = FALSE: REM ARC PHASE (ALTERNATES EACH STEP): TRUE=BELOW AXIS, FALSE=ABOVE AXIS
9260 S = 0: REM STEP SIZE (INITIALLY 0)
9300  REM SCREEN PARAMETERS
9310 XZ = 256:YZ = 192: REM SCREEN SIZE IN PIXELS
9320 XA = 0:YA = 0:XB = XZ - 1:YB = YZ - 1: REM SCREEN BOUNDARIES
9330 SC = 2: REM SCALE FACTOR: PIXELS PER STEP UNIT
9340 D =  SQR (XZ * XZ + YZ * YZ): REM SCREEN DIAGONAL LENGTH
9350 IS =  INT ((D - 1) / SC): REM MAX IN-SCREEN POSITION
9360 AA =  ATN (YZ / XZ): REM AXIS ANGLE = SCREEN DIAGONAL ANGLE
9370 CA =  COS (AA):SA =  SIN (AA): REM COSINE & SINE OF AXIS ANGLE
9380 SX = SC * CA:SY = SC * SA: REM HORIZONTAL SCALE FACTOR, VERTICAL SCALE FACTOR
9400  REM INITIALIZE SCREEN
9410  rem : REM GRAPHIC MODE 256X192 BLACK & WHITE
9420  line (XA,YB)-(XB,YA): REM DRAW AXIS: SCREEN DIAGONAL FROM BOTTOM-LEFT TO UPPER-RIGHT
9999  RETURN
